lab12=# -- My example 2: sales performance of each seller (CTE + CASE + LEFT JOIN)
lab12=# -- Total sales per employee, compared with the team average.
lab12=# WITH seller_totals AS (
lab12(#     SELECT e.name,
lab12(#            COALESCE(SUM(s.amount), 0) AS total_sales
lab12(#     FROM employees e
lab12(#     LEFT JOIN sales s ON s.employee_id = e.id
lab12(#     WHERE e.department IN ('Sales', 'Marketing')
lab12(#     GROUP BY e.name
lab12(# ),
lab12-# team_average AS (
lab12(#     SELECT AVG(total_sales) AS avg_sales FROM seller_totals
lab12(# )
lab12-# SELECT st.name,
lab12-#        st.total_sales,
lab12-#        ROUND(ta.avg_sales, 2) AS team_avg,
lab12-#        CASE
lab12-#            WHEN st.total_sales > ta.avg_sales THEN 'Above target'
lab12-#            ELSE 'Below target'
lab12-#        END AS performance
lab12-# FROM seller_totals st
lab12-# CROSS JOIN team_average ta
lab12-# ORDER BY st.total_sales DESC;
       name        | total_sales | team_avg | performance
-------------------+-------------+----------+--------------
 Bermet Asanova    |     3200.00 |  1933.33 | Above target
 Aibek Toktomushev |     1900.00 |  1933.33 | Below target
