lab15=# -- My example 1: ABC analysis of products (CTE + window functions + CASE)
lab15=# -- A = products that bring the first 80% of revenue, B = next 15%, C = the rest
lab15=# WITH product_revenue AS (
lab15(#     SELECT product_name, SUM(amount) AS revenue
lab15(#     FROM orders
lab15(#     GROUP BY product_name
lab15(# ),
lab15-# cumulative AS (
lab15(#     SELECT product_name, revenue,
lab15(#            ROUND(100 * SUM(revenue) OVER (ORDER BY revenue DESC)
lab15(#                      / SUM(revenue) OVER (), 1) AS cumulative_pct
lab15(#     FROM product_revenue
lab15(# )
lab15-# SELECT product_name, revenue, cumulative_pct,
lab15-#        CASE
lab15-#            WHEN cumulative_pct <= 80 THEN 'A'
lab15-#            WHEN cumulative_pct <= 95 THEN 'B'
lab15-#            ELSE 'C'
lab15-#        END AS abc_class
lab15-# FROM cumulative
lab15-# ORDER BY revenue DESC;

lab15=# -- My example 2: month-over-month revenue growth per store (CTE + LAG)
lab15=# WITH monthly AS (
lab15(#     SELECT store,
lab15(#            DATE_TRUNC('month', sale_date)::DATE AS month,
lab15(#            SUM(revenue) AS revenue
lab15(#     FROM store_sales

lab15(# )
lab15-# SELECT store, month, revenue,
lab15-#        LAG(revenue) OVER (PARTITION BY store ORDER BY month) AS prev_month,
